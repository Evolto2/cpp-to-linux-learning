cmake_minimum_required(VERSION 3.20)
project(cpp_to_linux_learning LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(calculator projects/01-calculator/upstream/Console_calculator_program.cpp)
add_executable(student_manager projects/02-student-manager/upstream/main.cpp)
find_package(Threads REQUIRED)
set(ASIO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/projects/03-tcp-echo/upstream")
foreach(role server client)
  add_executable(tcp_echo_${role} "${ASIO_ROOT}/src/examples/cpp11/echo/blocking_tcp_echo_${role}.cpp")
  target_include_directories(tcp_echo_${role} SYSTEM PRIVATE "${ASIO_ROOT}/include")
  target_compile_definitions(tcp_echo_${role} PRIVATE ASIO_STANDALONE)
  target_link_libraries(tcp_echo_${role} PRIVATE Threads::Threads)
  if(WIN32)
    target_compile_definitions(tcp_echo_${role} PRIVATE _WIN32_WINNT=0x0601)
    target_link_libraries(tcp_echo_${role} PRIVATE ws2_32 mswsock)
  endif()
endforeach()
foreach(target calculator student_manager tcp_echo_server tcp_echo_client)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
